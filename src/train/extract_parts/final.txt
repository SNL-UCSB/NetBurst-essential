def get_final_representations():
    """
    Extract final representations from all data points and save to CSV.
    Uses the same data loading format as main() but outputs embeddings instead of training.
    """
    parser = argparse.ArgumentParser()
    parser.add_argument("parquet_root", help="Root path to parquet data")
    parser.add_argument("--model", default="amazon/chronos-t5-small", 
                        help="Path to trained model directory")
    parser.add_argument("--batch_size", type=int, default=16)
    parser.add_argument("--limit", type=int, default=None)
    parser.add_argument("--max_len", type=int, default=None, 
                        help="Maximum length of series to consider")
    parser.add_argument("--nonhierarchical", action="store_true",
                        help="If set, use non-hierarchical data loading")
    parser.add_argument("--output_csv", default="final_representations.csv",
                        help="Output CSV file path")
    parser.add_argument("--isCSV", type=int, default=None)
    parser.add_argument("--outbound_only", action="store_true",
                        help="Use outbound data only for hierarchical loading")
    args = parser.parse_args()

    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    print(f"Using device: {device}")

    # Load data using the same format as main()
    if args.isCSV:
        # Load from CSV
        df = pd.read_csv(args.parquet_root)
        all_series = df.values.tolist()
    else:
        all_series = load_cleaned_series(
            args.parquet_root,
            min_len=10,
            max_len=args.max_len,
            limit=args.limit,
            nonHierarchical=args.nonhierarchical,
            outbound_only=args.outbound_only
        )

    print(f"Loaded {len(all_series)} series")

    # Load the trained model
    print("Loading trained model...")
    
    # Load boundaries from the model directory
    with open(os.path.join(args.model, "boundaries.pkl"), "rb") as f:
        boundaries = pickle.load(f)

    # Create the predictor with trained weights
    predictor = ChronosBinPredictor(
        boundaries, 
        "<data-root>/models--amazon--chronos-t5-small/snapshots/476a71b73e6205f7987e811a81f355b9791c9256", 
        device
    ).to(device)

    # Load the fine-tuned weights
    state = torch.load(os.path.join(args.model, "chronos_best.pt"), map_location=device)
    state = strip_prefix_if_present(state, ["model.model.", "model."])
    predictor.model.model.load_state_dict(state, strict=True)
    predictor.model.eval()

    print("Model loaded successfully")

    # Process data in batches to extract representations
    all_representations = []
    series_ids = []
    tokenizer_state = None
    
    print("Extracting representations...")
    
    with torch.no_grad():
        for i in range(0, len(all_series), args.batch_size):
            batch_series = all_series[i:i+args.batch_size]
            
            try:
                # Convert to numpy array format as expected by pipeline.embed()
                batch_data = []
                for series in batch_series:
                    if isinstance(series, list):
                        batch_data.append(np.array(series, dtype=np.float32))
                    else:
                        batch_data.append(np.array(series, dtype=np.float32))
                
                # Convert to tensor format
                batch_np = np.array(batch_data, dtype=np.float32)
                batch_tensor = torch.from_numpy(batch_np).to(device)
                
                # Use pipeline.embed() method like in the reference implementation
                batch_emb, tokenizer_state = predictor.pipeline.embed(
                    batch_tensor,
                )
                
                # Extract the last token embedding like in the reference: batch_emb[:,-1,:]
                # This gives us the final representation for each series
                representations = batch_emb[:, -1, :].detach().cpu().numpy()
                
                all_representations.extend(representations)
                
                # Store series IDs
                for j in range(len(batch_series)):
                    series_ids.append(i + j)
                
                if (i // args.batch_size + 1) % 10 == 0:
                    print(f"Processed {i + len(batch_series)}/{len(all_series)} series")
                    
            except Exception as e:
                print(f"Error processing batch {i//args.batch_size}: {e}")
                # Skip this batch and continue
                continue

    # Convert to DataFrame and save to CSV
    print("Saving representations to CSV...")
    
    # Convert list of numpy arrays to 2D array
    representations_array = np.array(all_representations)
    
    # Create column names for the representation dimensions
    n_dims = representations_array.shape[1]
    columns = ['series_id'] + [f'dim_{i}' for i in range(n_dims)]
    
    # Create DataFrame
    df_representations = pd.DataFrame()
    df_representations['series_id'] = series_ids
    
    # Add representation dimensions
    for i in range(n_dims):
        df_representations[f'dim_{i}'] = representations_array[:, i]
    
    # Save to CSV
    df_representations.to_csv(args.output_csv, index=False)
    
    print(f"Saved {len(representations_array)} representations to {args.output_csv}")
    print(f"Representation shape: {representations_array.shape}")
    print(f"CSV columns: {list(df_representations.columns)}")
